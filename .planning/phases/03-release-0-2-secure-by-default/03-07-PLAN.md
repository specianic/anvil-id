```rust
pub enum RouteClass { Auth, Token, Admin, Default }
pub enum LimitKey {
    Ip(IpAddr),
    Identifier { tenant: TenantId, ident: [u8; 32] },          // SHA-256 of the normalised identifier (callers hash)
    IdentIp { tenant: TenantId, ident: [u8; 32], ip: IpAddr },
    Client { tenant: TenantId, client_id: String },
    User { tenant: TenantId, user: UserId },                    // second-factor attempts
}
pub enum Decision { Allow, Deny { retry_after: Duration }, RequireCaptcha }
pub struct Backoff { pub failures: u32, pub locked_for: Option<Duration> }
#[async_trait] pub trait RateLimiter: Send + Sync {
    async fn check(&self, class: RouteClass, key: &LimitKey) -> Decision;            // GCRA request rate
    async fn backoff(&self, key: &LimitKey) -> Option<Duration>;                      // remaining lock window
    async fn record_failure(&self, key: &LimitKey) -> Backoff;
    async fn record_success(&self, key: &LimitKey);
    async fn brute_force_gate(&self, strict: &LimitKey, loose: &LimitKey, captcha_available: bool) -> Decision;
}
pub struct LimiterConfig { per-class Quota (per_second/burst), free_attempts, base, cap, loose_threshold_per_hour, enabled }
impl Default for LimiterConfig  // auth 10/min burst 5 per IP; token 50/s per IP + 100/s per client; admin 20/s; default 50/s; free 5; base 1 s; cap 15 min; loose 50/h
```
anvil-cache may depend on anvil-core (Clock, ids) and tokio; never on storage crates (scripts/check-deps.sh).
