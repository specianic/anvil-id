pub fn generate_rs256() -> Result<GeneratedKey, JoseError>;               // RSA-2048, kid = thumbprint, use=sig, alg=RS256

impl SigningKey { pub fn from_pkcs8(kid: &str, alg: Alg, pkcs8: &[u8]) -> Result<Self, JoseError>;
  pub fn kid(&self) -> &str;
  pub fn sign_jwt<C: Serialize>(&self, typ: Option<&str>, claims: &C) -> Result<String, JoseError>; } // header {alg,kid,typ?}

pub fn verify_client_assertion(assertion: &str, client_jwks: &Jwks, rules: &AssertionRules) -> Result<VerifiedAssertion, JoseError>;
